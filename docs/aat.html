aat

aat: a reductive Agent-Action-Target (AAT) model of natural-language syntax (see aat-model.md), plus a DSPy-based pipeline applying it to English (aat.english).

This top-level module re-exports only aat.core -- the language-agnostic model itself -- so import aat never requires dspy to be installed. English-specific application code lives in aat.english and is imported separately:

from aat import CitableToken, AATNode, AATGraph, graph_to_mermaid   # always available
from aat.english import analyze_passage                              # needs the 'english' extra

See README.md for the reasoning behind this split, and USAGE.md for worked examples of both.

24class AATNode(BaseModel):
25    """One node in an AAT graph.
26
27    `context` and `id` identify the token this node is built from -- for a
28    single-token action, an agent, or a target, `id` is just that token's
29    own id. For a *compound* action (aat-model.md's "Actions" section --
30    e.g. "was eating"), `id` is instead the id of the *principal verb*
31    token within the compound (e.g. the token for "eating"), and `value`
32    is the space-joined text of every component token in surface order
33    (e.g. "was eating"), not just the principal verb's own text.

36      - `None`, for an *independent* action node (aat-model.md's
37        "Actions" section);

39        of the governing action it depends on);
40      - another action node's `id`, for an agent or target node (the id
41        of the action it's the agent/target of) -- agent and target nodes
42        always have a related_node; only an action node's related_node
43        can be `None`.

def validate( tokens: List[CitableToken], graph: AATGraph) -> List[str]: 
16def validate(tokens: List[CitableToken], graph: AATGraph) -> List[str]:
17    """Return a list of human-readable problem descriptions found in
18    `graph`, given the `tokens` it was built from. An empty list means no
19    referential problems were found (not that the analysis is correct).
20
21    Checks:
22      - every node's (context, id) matches a real token in `tokens`;
23      - every node's `value` is non-empty;
24      - an action node's `related_node`, if set, resolves to another
25        action node's id in the same context;
26      - an agent or target node's `related_node` is set (not None) and
27        resolves to an action node's id in the same context.
28    """
29    problems: List[str] = []
30
31    token_ids = {(t.context, t.id) for t in tokens}
32    action_ids = {(n.context, n.id) for n in graph.nodes if n.role == "action"}
33
34    for node in graph.nodes:
35        if (node.context, node.id) not in token_ids:
36            problems.append(
37                f"node ({node.context!r}, {node.id!r}, role={node.role!r}) "
38                "does not match any token in the input token list"
39            )
40        if not node.value:
41            problems.append(
42                f"node ({node.context!r}, {node.id!r}) has an empty value"
43            )
44
45        if node.role == "action":
46            if node.related_node is not None and (node.context, node.related_node) not in action_ids:
47                problems.append(
48                    f"action node ({node.context!r}, {node.id!r}) has related_node "
49                    f"{node.related_node!r}, which is not the id of any action node "
50                    "in this context"
51                )
52        else:  # agent or target
53            if node.related_node is None:
54                problems.append(
55                    f"{node.role} node ({node.context!r}, {node.id!r}) has no "
56                    "related_node -- every agent/target node must relate to an "
57                    "action node"
58                )
59            elif (node.context, node.related_node) not in action_ids:
60                problems.append(
61                    f"{node.role} node ({node.context!r}, {node.id!r}) has related_node "
62                    f"{node.related_node!r}, which is not the id of any action node "
63                    "in this context"
64                )
65
66    return problems

def graph_to_mermaid( graph: AATGraph, orientation: str = 'BT', color_by_action: bool = True) -> Tuple[str, List[str]]: 
 93def graph_to_mermaid(
 94    graph: AATGraph,
 95    orientation: str = "BT",
 96    color_by_action: bool = True,
 97) -> Tuple[str, List[str]]:
 98    """Build a Mermaid `graph` diagram from an AATGraph.
 99
100    `orientation` is Mermaid's own flowchart direction code -- `BT`
101    (bottom-to-top, the default here), `TB`/`TD` (top-down -- synonyms),
102    `LR`, or `RL` -- used verbatim (uppercased) in the diagram's opening
103    line (`graph BT`, etc.). Matched case-insensitively against
104    `_VALID_ORIENTATIONS`; anything else raises `ValueError` naming the
105    valid options, rather than silently producing invalid Mermaid syntax.
106    See https://mermaid.js.org/syntax/flowchart.html#direction.
107
108    `color_by_action` (default True) -- see this module's own docstring.
109    Pass False for a plain, uncolored diagram.
110
111    Returns (diagram_text, warnings). A node whose `related_node` doesn't
112    resolve to any node actually present in `graph` (same context) is
113    still drawn, but its edge is skipped and reported as a warning --
114    normally a sign `graph` failed aat.core.validate.validate() upstream,
115    worth checking there first. If `color_by_action` is True and the graph
116    has more distinct actions than the palette has colors (currently 8),
117    one warning notes that colors repeat.
118    """
119    orientation = _normalize_orientation(orientation)
120
121    by_key: Dict[Tuple[str, str], AATNode] = {_node_key(n): n for n in graph.nodes}
122
123    lines = [f"graph {orientation}"]
124    for node in graph.nodes:
125        open_b, close_b = _ROLE_BRACKETS.get(node.role, ("[", "]"))
126        lines.append(f'    {node.id}{open_b}"{_escape_label(node.value)}"{close_b}')
127
128    warnings: List[str] = []
129    for node in graph.nodes:
130        if node.related_node is None:
131            continue
132        target_key = (node.context, node.related_node)
133        if target_key not in by_key:
134            warnings.append(
135                f"skipped edge {node.id} -[{node.role}]-> {node.related_node}: "
136                "target is not a node in this graph"
137            )
138            continue
139        edge_label = "dependent" if node.role == "action" else node.role
140        lines.append(f"    {node.id} -->|{edge_label}| {node.related_node}")
141
142    if color_by_action:
143        color_of_node, color_warnings = assign_action_colors(graph)
144        warnings.extend(color_warnings)
145
146        if color_of_node:
147            lines.append("")
148
149            # Group by color VALUE, not by action -- two actions that
150            # cycle to the same palette color (more actions than the
151            # palette has slots) share a single classDef/class pair here
152            # rather than emitting two redundant classes with identical
153            # colors.
154            colors_in_order: List[ColorTriple] = []
155            seen_colors = set()
156            for node in graph.nodes:
157                color = color_of_node.get(_node_key(node))
158                if color is not None and color not in seen_colors:
159                    seen_colors.add(color)
160                    colors_in_order.append(color)
161
162            class_name_of_color: Dict[ColorTriple, str] = {
163                color: f"c{i}" for i, color in enumerate(colors_in_order)
164            }
165            for color in colors_in_order:
166                fill, stroke, text = color
167                lines.append(
168                    f"    classDef {class_name_of_color[color]} "
169                    f"fill:{fill},stroke:{stroke},color:{text};"
170                )
171
172            nodes_by_class: Dict[str, List[str]] = {}
173            for node in graph.nodes:
174                color = color_of_node.get(_node_key(node))
175                if color is None:
176                    continue
177                class_name = class_name_of_color[color]
178                nodes_by_class.setdefault(class_name, []).append(node.id)
179
180            for class_name, node_ids in nodes_by_class.items():
181                lines.append(f"    class {','.join(node_ids)} {class_name};")
182
183    return "\n".join(lines), warnings

def assign_action_colors( graph: AATGraph) -> Tuple[Dict[Tuple[str, str], Tuple[str, str, str]], List[str]]: 
48def assign_action_colors(graph: AATGraph) -> Tuple[Dict[NodeKey, ColorTriple], List[str]]:
49    """Return (color_of_node, warnings).
50
51    `color_of_node` maps every node's (context, id) to a (fill, stroke,
52    text) hex triple, for every node that clusters with some action: an
53    action node itself (dependent or not -- see this module's own
54    docstring), or an agent/target node whose `related_node` resolves to
55    an action node in the same context. A node with no resolvable cluster
56    (e.g. an agent/target with a broken related_node -- shouldn't happen
57    in a graph that's passed aat.core.validate.validate()) is simply
58    absent from the mapping.
59
60    Colors are assigned to actions in the order they first appear in
61    `graph.nodes`, cycling through the palette (currently 8 colors) if
62    there are more distinct actions than that -- in which case two
63    distinct actions (and their own agent/target nodes) end up sharing the
64    identical color. `warnings` has one entry naming this when it happens;
65    otherwise it's empty.
66    """
67    actions = [n for n in graph.nodes if n.role == "action"]
68
69    warnings: List[str] = []
70    if len(actions) > len(_ACTION_COLOR_PALETTE):
71        warnings.append(
72            f"{len(actions)} actions but only {len(_ACTION_COLOR_PALETTE)} palette "
73            "colors -- colors repeat and some distinct actions will share a color"
74        )
75
76    action_index: Dict[NodeKey, int] = {_node_key(a): i for i, a in enumerate(actions)}
77
78    by_key: Dict[NodeKey, AATNode] = {_node_key(n): n for n in graph.nodes}
79    cluster_of_node: Dict[NodeKey, NodeKey] = {}
80    for node in graph.nodes:
81        key = _node_key(node)
82        if node.role == "action":
83            cluster_of_node[key] = key
84        elif node.related_node is not None:
85            related_key = (node.context, node.related_node)
86            if related_key in by_key:
87                cluster_of_node[key] = related_key
88
89    color_of_node: Dict[NodeKey, ColorTriple] = {}
90    for node in graph.nodes:
91        key = _node_key(node)
92        cluster_key = cluster_of_node.get(key)
93        if cluster_key is None or cluster_key not in action_index:
94            continue
95        color_of_node[key] = _ACTION_COLOR_PALETTE[action_index[cluster_key] % len(_ACTION_COLOR_PALETTE)]
96
97    return color_of_node, warnings
