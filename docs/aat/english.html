29def tokenize(passage: CitedPassage) -> List[CitableToken]:
30    """Tokenize `passage.text` into a list of CitableToken, with ids 't1',

33
34    Splits on whitespace, keeps an apostrophe-joined contraction or

def analyze_passages( passages: List[aat.CitedPassage]) -> Tuple[List[aat.CitableToken], aat.AATGraph]: 
16def analyze_passages(passages: List[CitedPassage]) -> Tuple[List[CitableToken], AATGraph]:
17    """Tokenize and analyze every passage in `passages`, independently
18    (each passage's tokens are numbered from 't1' within its own context
19    -- see CitableToken's docstring), and return (tokens, graph): `tokens`
20    is every passage's tokens concatenated in order, `graph` is one
21    AATGraph combining every passage's nodes.
22
23    Prints a warning for any passage whose analysis fails validate() (a
24    referential problem -- see aat.core.validate.validate); it does not
25    raise, since a referential problem is a sign the LM's output needs a
26    re-run or a prompt tweak, not necessarily that the caller's own code
27    is broken.
28    """
29    all_tokens: List[CitableToken] = []
30    all_nodes: List[AATNode] = []
31
32    for passage in passages:
33        tokens = tokenize(passage)
34        result = analyze(passage=passage.text, tokens=tokens)
35
36        problems = validate(tokens, result)
37        if problems:
38            print(f"Validation warnings (context {passage.context!r}):")
39            for p in problems:
40                print(f"  - {p}")
41
42        all_tokens.extend(tokens)
43        all_nodes.extend(result.nodes)
44
45    return all_tokens, AATGraph(nodes=all_nodes)

 32def aat_metric(
 33    gold: "dspy.Example",
 34    pred: "dspy.Prediction",
 35    trace: Optional[object] = None,
 36    pred_name: Optional[str] = None,
 37    pred_trace: Optional[object] = None,
 38    program_trace: Optional[object] = None,
 39) -> "dspy.Prediction":
 40    """Score a prediction's `nodes` against `gold.nodes` (both lists of
 41    AATNode). Returns dspy.Prediction(score=..., feedback=...) -- GEPA's
 42    expected "ScoreWithFeedback" shape. `pred_name`/`pred_trace` are
 43    accepted for protocol compatibility (GEPA can call a metric either at
 44    the program level or per-predictor level) but not used to change
 45    scoring -- there's only one predictor in `analyze`.
 46
 47    `score` is in [0, 1]: the fraction of (gold-node-count +
 48    extra-predicted-node-count) correctly accounted for, where a gold
 49    node counts as correct only if the prediction has a node at the same
 50    (context, id, role) with a matching `value` and `related_node`.
 51    `feedback` names every missing, extra, or mismatched node.
 52    """
 53    problems = []
 54
 55    gold_nodes = {_key(n): n for n in gold.nodes}
 56    pred_nodes_list = list(getattr(pred, "nodes", None) or [])
 57    pred_nodes = {}
 58    for n in pred_nodes_list:
 59        k = _key(n)
 60        if k in pred_nodes:
 61            problems.append(
 62                f"nodes has more than one entry for {k} (only the first is scored)"
 63            )
 64            continue
 65        pred_nodes[k] = n
 66
 67    correct = 0
 68    total = len(gold_nodes)
 69
 70    for key, g in gold_nodes.items():
 71        context, id_, role = key
 72        if key not in pred_nodes:
 73            problems.append(f"missing {role} node ({context!r}, {id_!r}) -- value {g.value!r}")
 74            continue
 75        p = pred_nodes[key]
 76        ok = True
 77        if p.value != g.value:
 78            problems.append(
 79                f"{role} node ({context!r}, {id_!r}): expected value={g.value!r}, got {p.value!r}"
 80            )
 81            ok = False
 82        if p.related_node != g.related_node:
 83            problems.append(
 84                f"{role} node ({context!r}, {id_!r}): expected related_node={g.related_node!r}, "
 85                f"got {p.related_node!r}"
 86            )
 87            ok = False
 88        if ok:
 89            correct += 1
 90
 91    extra_keys = set(pred_nodes) - set(gold_nodes)
 92    for context, id_, role in sorted(extra_keys):
 93        problems.append(
 94            f"unexpected extra {role} node ({context!r}, {id_!r}) not in the gold answer"
 95        )
 96    total += len(extra_keys)
 97
 98    score = correct / total if total else 1.0
 99
100    if not problems:
101        feedback = (
102            "Perfect match with the gold analysis: every node's role, value, "
103            "and related_node are correct."
104        )
105    else:
106        feedback = (
107            f"Score {score:.2f} ({correct}/{total} nodes correct). Problems found:\n- "
108            + "\n- ".join(problems)
109        )
110
111    return dspy.Prediction(score=score, feedback=feedback)

109def tokens_to_html(tokens: List[CitableToken], graph: Optional[AATGraph] = None) -> str:
110    """Render `tokens` as one continuous HTML string: ordinary words get
111    a space before them, punctuation attaches per this module's own
112    docstring (left-joining by default; opening brackets and the first
113    of a quote pair right-join instead).
114
115    If `graph` is given, every token whose (context, id) matches a node
116    in `graph` is highlighted: background/text colored the same way
117    that node is colored in `aat.core.mermaid.graph_to_mermaid()`'s
118    diagram (via `aat.core.coloring.assign_action_colors()`), with a
119    role-specific border -- a box around an `action` token, a rounded
120    box around an `agent` token, an underline under a `target` token.
121    Omit `graph` (or pass `None`) for plain, unhighlighted text.
122
123    Every token's text is HTML-escaped before being emitted, span or
124    not -- real passage text can contain a literal `<`, `>`, `&`, or
125    quote character, which would otherwise be indistinguishable from
126    markup to anything that re-parses this output.
127    """
128    color_of_node: Dict[Tuple[str, str], Tuple[str, str, str]] = {}
129    role_of_node: Dict[Tuple[str, str], str] = {}
130    if graph is not None:
131        color_of_node, _warnings = assign_action_colors(graph)
132        role_of_node = {(n.context, n.id): n.role for n in graph.nodes}
133
134    quote_counts: Dict[str, int] = {}
135    pieces: List[str] = []
136    previous_class: Optional[str] = None
137
138    for token in tokens:
139        cls = _classify(token, quote_counts)
140        rendered = html.escape(token.value)
141
142        key = (token.context, token.id)
143        color = color_of_node.get(key)
144        if color is not None:
145            fill, stroke, text_color = color
146            role = role_of_node.get(key, "")
147            role_style = _ROLE_STYLE.get(role, "").format(stroke=stroke)
148            rendered = (
149                f'<span style="background-color: {fill}; color: {text_color}; {role_style}">'
150                f"{rendered}</span>"
151            )
152
153        if not pieces:
154            # Nothing precedes the first token -- never prepend a space.
155            pieces.append(rendered)
156        elif cls == _LEFT:
157            pieces.append(rendered)
158        elif cls == _RIGHT:
159            pieces.append(" " + rendered)
160        else:  # _NORMAL
161            if previous_class == _RIGHT:
162                pieces.append(rendered)
163            else:
164                pieces.append(" " + rendered)
165
166        previous_class = cls
167
168    return "".join(pieces)
